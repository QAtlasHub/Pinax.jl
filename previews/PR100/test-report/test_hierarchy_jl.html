Sec. 118. numbering=:part + a :page numberer stamps per-part page badges

nb(kind, c) =
    if kind === :page
        (c.part_id === :thermal ? "EQ$(c.page)" : "GQ$(c.page)")
    elseif kind === :figure
        "Fig. $(c.figure)"
    else
        "($(c.equation))"
    end
Pinax.reset!(; numbering=:part, numberer=nb)
Pinax.render(; out=sitedir("num"))
